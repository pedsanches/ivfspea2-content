% Incorpora uma página PDF externa em tamanho A4 sem depender de pdfpages,
% que não integra o pacote Tectonic usado pelo projeto.
\newcommand{\incluirpaginaexterna}[2][1]{%
  \clearpage
  \thispagestyle{empty}%
  \begingroup
  \noindent\makebox[0pt][l]{%
    \ifodd\value{page}%
      \hspace*{\dimexpr-1in-\hoffset-\oddsidemargin\relax}%
    \else
      \hspace*{\dimexpr-1in-\hoffset-\evensidemargin\relax}%
    \fi
    \raisebox{\dimexpr1in+\voffset+\topmargin+\headheight+\headsep+\topskip-\paperheight\relax}[0pt][0pt]{%
      \includegraphics[page=#1,width=\paperwidth,height=\paperheight]{#2}}}%
  \null
  \endgroup
  \clearpage
}
% Posicionamento de floats. Os capítulos de resultados carregam 18 tabelas e 6
% figuras, e com os padrões do LaTeX o acúmulo produzia uma página só de floats
% ("Text page N contains only floats"). Estes valores dão mais espaço a floats
% no topo e exigem um mínimo de texto por página, mantendo cada float perto do
% parágrafo que o discute.
\renewcommand{\topfraction}{0.85}
\renewcommand{\bottomfraction}{0.70}
\renewcommand{\textfraction}{0.15}
\renewcommand{\floatpagefraction}{0.75}
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{4}

% Tabelas. Só filetes horizontais (booktabs), sem filete vertical: a tabela fica
% aberta nas laterais, como na apresentação tabular do IBGE adotada pela ABNT.
% Toda tabela ocupa a largura do texto: tabularx quando há coluna de texto com
% quebra de linha (tipo L), tabular* nas demais. O corpo é \small e a nota,
% \footnotesize; nenhuma tabela é escalada. Como a legenda fica acima da tabela,
% position=top põe o espaço da legenda entre ela e o filete superior.
\usepackage{booktabs}
\usepackage{tabularx}
% Centraliza no eixo vertical os rótulos que agrupam várias linhas.
\usepackage{multirow}
\newcolumntype{L}{>{\raggedright\arraybackslash}X}
\captionsetup[table]{position=top}
% \FloatBarrier impede que tabelas de uma subseção sejam empurradas para depois
% da subseção seguinte (Tabelas de HV por instância antes da Síntese, R19).
