% Incorpora uma página PDF externa em tamanho A4 sem depender de pdfpages,
% que não integra o pacote Tectonic usado pelo projeto.
\newcommand{\incluirpaginaexterna}[2][1]{%
  \clearpage
  \thispagestyle{empty}%
  \begingroup
  \noindent\makebox[0pt][l]{%
    \ifodd\value{page}%
      \hspace*{\dimexpr-1in-\hoffset-\oddsidemargin\relax}%
    \else
      \hspace*{\dimexpr-1in-\hoffset-\evensidemargin\relax}%
    \fi
    \raisebox{\dimexpr1in+\voffset+\topmargin+\headheight+\headsep+\topskip-\paperheight\relax}[0pt][0pt]{%
      \includegraphics[page=#1,width=\paperwidth,height=\paperheight]{#2}}}%
  \null
  \endgroup
  \clearpage
}
% Posicionamento de floats. Os capítulos de resultados carregam 18 tabelas e 6
% figuras, e com os padrões do LaTeX o acúmulo produzia uma página só de floats
% ("Text page N contains only floats"). Estes valores dão mais espaço a floats

% parágrafo que o discute.
